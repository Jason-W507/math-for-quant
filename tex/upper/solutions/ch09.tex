\chapter{第 9 章分层答案}
\label{app:ch09-solutions}

\section*{口述概念}
\begin{enumerate}[leftmargin=*]
  \item 几乎处处收敛要求除零测集外逐点收敛；依概率收敛要求每个固定误差阈值的失败概率趋零；$L^p$ 收敛要求 $p$ 次误差矩趋零；分布收敛要求分布函数在极限分布连续点收敛。前三者需要共同概率空间，分布收敛只涉及概率律。
  \item 有效箭头是 $a.s.\Rightarrow\mathbb P\Rightarrow d$、$L^p\Rightarrow\mathbb P$，以及概率空间上 $q\ge p$ 时 $L^q\Rightarrow L^p$。依概率只压低超阈事件概率，稀有但巨大的值仍可保持 $L^1$ 损失；一致可积性恰好排除这种尾部逃逸。
  \item WLLN 给样本均值依概率趋近总体均值，SLLN 给几乎处处收敛；CLT 给中心化缩放和的极限形状；Berry--Esseen 给正态近似的有限样本 sup 误差；集中界无需先取极限，直接上界当前 $n$ 的尾概率。
\end{enumerate}

\section*{笔试推导}
\begin{enumerate}[leftmargin=*]
  \item 对任意 $\varepsilon>0$，
  \[
    \mathbb P(|X_n-X|>\varepsilon)
    \le \varepsilon^{-p}\mathbb E|X_n-X|^p\to0.
  \]
  若 $X_n\to X$ 依概率，从任意子列递归选 $n_k$ 使
  $\mathbb P(|X_{n_k}-X|>2^{-k})\le2^{-k}$；概率和有限，Borel--Cantelli 给进一步子列几乎处处收敛。反向若依概率失败，则存在子列、$\varepsilon,\delta>0$ 使偏差概率始终至少为 $\delta$，不可能含几乎处处收敛的进一步子列。
  \item iid 时
  $\mathbb E\bar X_n=\mu$、$\operatorname{Var}(\bar X_n)=\sigma^2/n$，故 Chebyshev 给
  $\mathbb P(|\bar X_n-\mu|\ge\varepsilon)\le\sigma^2/(n\varepsilon^2)$。Bernoulli$(0.3)$ 有 $\sigma^2=0.21$，$n=200$ 时标准误
  $\sqrt{0.21/200}=0.0324037$。取 $t=0.1$，Chebyshev、Hoeffding、Bernstein 界依次为
  \[
    0.105,\qquad 2e^{-4}=0.0366313,\qquad
    2\exp\{-2/(0.42+0.066\overline6)\}=0.0328290.
  \]
  \item 若 $\mathbb EX=0$、$\operatorname{Var}(X)=\sigma^2$，则特征函数
  $\varphi(u)=1-\sigma^2u^2/2+o(u^2)$。标准化和的特征函数为
  $[\varphi(t/(\sigma\sqrt n))]^n\to e^{-t^2/2}$，Lévy 定理完成证明。Bernoulli 的三阶绝对中心矩为
  $\rho_3=p(1-p)^3+(1-p)p^3=0.1218$；取 $C=0.56$ 得界
  $C\rho_3/((0.21)^{3/2}\sqrt{200})=0.0501177$。三角阵列中若每行只有一个方差为 $1$ 的 Rademacher 项，则 $s_n=1$ 且任意 $\varepsilon<1$ 的 Lindeberg 比率恒为 $1$，所以标准化和不会变成正态。
\end{enumerate}

\section*{数值编程}
\begin{enumerate}[leftmargin=*]
  \item 有限方差独立样本的标准误为 $\sqrt{p(1-p)/n}$，区间随 $n$ 大致按平方根缩窄。
  \item 标准 Cauchy 平均仍服从标准 Cauchy，分位区间在分布上不缩窄；其均值与方差条件不成立。
  \item 对 $p=0.3,n=200,t=0.1$，精确尾概率约为 $0.00256512$，两界为 $0.105$ 与 $0.0366313$；保持同一非严格事件端点。
\end{enumerate}

\section*{研究判断}
\begin{enumerate}[leftmargin=*]
  \item 20 日重叠收益共享 19 日信息，不能把日数当独立样本量。应明确持有期为抽样单位，报告自相关与重叠结构；以非重叠子样本作基线，再比较 HAC、移动块/平稳块重采样及块长敏感性。若制度变化明显，还要分段或滚动验证，而非只改标准误。
  \item $p=0.01,n=10$ 时零成功概率为 $0.904382$，正态对称区间 $[-0.05167,0.07167]$ 越过参数空间，说明渐近形状尚未控制有限样本离散性。应报告精确二项区间或适合稀有事件的模型，并做覆盖率模拟、校准图和不同基准率压力测试。
\end{enumerate}
