\chapter{第 7 章分层答案}
\label{app:ch07-solutions}

\section*{口述概念}
\begin{enumerate}[leftmargin=*]
  \item $\Omega$ 列出状态，$\mathcal F$ 规定可提问事件，$\mathbb P$ 赋权；随机变量是可测映射，分布是推前测度。CDF 是累计概率，PMF 是点质量，密度是相对 Lebesgue 测度的导数。密度高度可以超过一，只要对全集积分为一；连续变量单点概率仍为零。
  \item LOTUS 把原空间上 $g(X)$ 的积分改写为对 $X$ 的律积分。若还需要 $Y=g(X)$ 的完整律，就用原像计算 CDF 或用 Jacobian 变换密度。非单调 $g$ 可能有多个原像，所有分支概率必须相加；离散平方账本中的质量合并就是最小例子。
  \item 同分布只比较边缘律；独立要求联合律等于边缘乘积；不相关只要求协方差为零且需二阶矩；IID 同时要求同分布和相互独立。独立推出不相关（矩存在时），反向不成立；同分布也不蕴含任何共同依赖结构。
\end{enumerate}

\section*{笔试推导}
\begin{enumerate}[leftmargin=*]
  \item 累加 $-1,0,2$ 的质量得到 $F(-1),F(0),F(1),F(2)=(1/4,3/4,3/4,1)$。期望为 $1/4$，二阶矩 $5/4$，方差为 $19/16$。$X^2$ 取 $0,1,4$，质量分别为 $1/2,1/4,1/4$；LOTUS 再给 $\mathbb E[X^2]=5/4$。
  \item 行和与列和都是 $(1/2,1/2)$。支持取 $(-1,1)$ 时两均值为零、方差为一，$\mathbb E[XY]=3/8+3/8-1/8-1/8=1/2$，故协方差与相关均为 $1/2$；与边缘外积的最大差为 $1/8$。若 $X$ 关于零对称且三阶矩存在，令 $Y=X^2$，则 $\operatorname{Cov}(X,Y)=\mathbb E[X^3]=0$，但 $Y$ 由 $X$ 决定。
  这里的差分变量满足：$D$ 与 $Z$ 均值为 $0$、方差为 $1$；四阶矩分别满足 $\mathbb E[D^4]=2$ 与 $\mathbb E[Z^4]=3$，而离散差分还给出 $\mathbb P(D=0)=1/2$。
  \item 若 $Y=e^Z$、$Z\sim N(\mu,\sigma^2)$，则
  \[
   f_Y(y)=\frac{1}{y\sigma\sqrt{2\pi}}
   \exp\!\left[-\frac{(\log y-\mu)^2}{2\sigma^2}\right],\quad y>0.
  \]
  由正态分布 MGF，$\mathbb E[Y^k]=e^{k\mu+k^2\sigma^2/2}$，代入 $k=1,2$ 得正文均值方差。Pareto$(\alpha)$ 的 $p$ 阶矩尾部为 $\alpha\int_1^\infty x^{p-\alpha-1}dx$，恰在 $p-\alpha<0$ 时收敛。
\end{enumerate}

\section*{数值编程}
\begin{enumerate}[leftmargin=*]
  \item oracle 不抽样，直接用固定质量向量求和。它核验
  \[
    (F(-1),F(0),F(1),F(2))=(0.25,0.75,0.75,1),
  \]
  平方推前在 $(0,1,4)$ 上的质量为 $(0.5,0.25,0.25)$；联合表独立性最大差为 $0.125$、协方差为 $0.5$。这些 expected 来自分数账本而非 NumPy 输出。
  \item 解析账本给出 $M_B(\log 2)=5/4$、$G_N(1/2)=e^{-1}$、$M_Z(1/2)=e^{1/8}$ 和 $\Phi(1)=0.841344746\ldots$；$U(-1,3)$ 的均值/方差为 $1,4/3$；$\mathrm{Exp}(2)$ 的均值和生存概率为 $1/2,e^{-2}$；Lognormal$(0,0.5^2)$ 的均值/方差为 $1.133148,0.364696$。Pareto 截断到 $10$ 与 $1000$ 时分别为 $2.302585$、$6.907755$。全部使用固定 $10^{-10}$ 绝对误差门。
  \item NaN expected 触发有限值门，删除正态分布 MGF 触发缺字段门，单元素边缘触发固定形状门，负质量触发非负门，更换支持值触发固定账本门，宽容差触发 tolerance 门，错误顶层 expected 触发 $1/4$ 发布标签门。恢复时只改临时 JSON，不得把 observed 写回 expected。
\end{enumerate}
