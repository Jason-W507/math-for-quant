\chapter{第 12 章分层答案}
\label{app:ch12-solutions}

\section*{口述概念}
\begin{enumerate}[leftmargin=*]
\item 严格平稳约束全部有限维分布，弱平稳只约束二阶矩；遍历使时间平均收敛到总体量。白噪声只要求不相关，IID 更强；创新还要求相对指定过去信息条件均值为零。
\item AR 根在单位圆外给因果表示，MA 根在单位圆外给可逆创新。ACF/PACF 的截尾规律是总体线索；有限样本受误差、近单位根和结构变化影响。
\item 预测在观测前，滤波用截至当前数据，平滑用全样本；创新似然累加一步预测误差密度。实时策略只能使用预测与滤波量，不能使用平滑值。
\end{enumerate}

\section*{笔试推导}
\begin{enumerate}[leftmargin=*,start=4]
\item 迭代 $m$ 次得 $X_t=\phi^mX_{t-m}+\sum_{j=0}^{m-1}\phi^j\varepsilon_{t-j}$。因 $|\phi|<1$ 且二阶矩有限，第一项在 $L^2$ 中消失，故得到 MA$(\infty)$ 表示。创新正交给
\[
\gamma(0)=\sigma^2\sum_{j\ge0}\phi^{2j}=\frac{\sigma^2}{1-\phi^2}.
\]
把方程乘 $X_{t-h}$ 并取期望；$h\ge1$ 时创新与过去正交，所以 $\gamma(h)=\phi\gamma(h-1)=\phi^h\gamma(0)$。再迭代未来值：
\[
X_{t+h}=\phi^hX_t+\sum_{j=0}^{h-1}\phi^j\varepsilon_{t+h-j}.
\]
第一项是条件期望，第二项由未来正交创新组成，故误差方差为 $\sigma^2\sum_{j=0}^{h-1}\phi^{2j}$。代入 $\phi=0.8,h=3,X_t=1$，本例理论方差为 $2.777778$，三步预测为 $0.512000$，预测误差方差为 $2.049600$。

\item 因 $X_t=(1-\phi L)^{-1}(1+\theta L)\varepsilon_t$，有 $\psi_0=1$，且对 $j\ge1$，$\psi_j=(\phi+\theta)\phi^{j-1}$。因此
\[
\gamma(0)=\sigma^2\left[1+(\phi+\theta)^2\sum_{j\ge0}\phi^{2j}\right]
=\sigma^2\frac{1+\theta^2+2\phi\theta}{1-\phi^2},
\]
\[
\gamma(1)=\sigma^2\sum_{j\ge0}\psi_j\psi_{j+1}
=\sigma^2\frac{(\phi+\theta)(1+\phi\theta)}{1-\phi^2}.
\]
代入 $(\phi,\theta,\sigma^2)=(0.6,0.3,1)$，ARMA 解析方差为 $2.265625$，$\rho(1)=0.732414$。方程 $1-0.6z=0$ 与 $1+0.3z=0$ 的根模为 $1.666667,3.333333$，均在单位圆外，分别通过因果与可逆检查。

\item 对 GARCH，条件标准化给 $\mathbb E(r_{t-1}^2)=\mathbb E(\sigma_{t-1}^2)=v$；取期望得 $v=\omega+(\alpha+\beta)v$，所以 $v=1$、持续性为 $0.9$。GARCH 半衰期为 $6.578813$（即 $\log(0.5)/\log(0.9)$）。对 $u_t=0.4u_{t-1}+e_t$ 两边减去 $u_{t-1}$，得到 $\Delta u_t=-0.6u_{t-1}+e_t$，即误差修正系数 $-0.6$。Kalman 每步有 $\nu_t=y_t-x_{t\mid t-1}$、$S_t=P_{t\mid t-1}+R$，故
\[
\ell_t=-\tfrac12\{\log(2\pi)+\log S_t+\nu_t^2/S_t\}.
\]
三步计算记录为
\[
\begin{array}{c|ccc}
t&1&2&3\\\hline
\nu_t&1&-1.055556&0.165385\\
S_t&2.25&1.805556&1.696154\\
\ell_t&-1.546626&-1.522920&-1.191183
\end{array}
\]
相加得到创新对数似然为 $-4.260729$。
\end{enumerate}

\section*{数值编程}
\begin{enumerate}[leftmargin=*]
  \item 均值为 $\phi^hX_t$，误差方差为 $\sigma^2\sum_{j=0}^{h-1}\phi^{2j}$；正态区间还需高斯创新或相应近似。
  \item $P^-=P+Q,K=P^-/(P^-+R),m=m^-+K(y-m^-),P=(1-K)P^-$。本章三观测例均值约为 $0.555556,0.084615,0.152494$。
  \item $Q$ 增大使预测更不确定，通常增大增益；$R$ 增大使观测更不可靠，降低增益。
\end{enumerate}

\section*{研究判断}
\begin{enumerate}[leftmargin=*,start=10]
\item 先审计特征可得时点和所有选择步骤，再做滚动/扩展窗口。随机切分 MSE 为 $1.436939$，时间切分 MSE 为 $2.226153$；前者更小是泄漏或非平稳警报。
\item Ljung--Box 不拒绝只说明所检滞后没有足够线性相关证据；还需检查平方残差、尾部、参数稳定性、真实预测损失和多模型选择偏差。AIC 也不验证交易成本或因果机制。
\item 在训练窗口内选配对与阶数，冻结后滚动估计价差和 GARCH；报告协整破裂、半衰期漂移、残差诊断、换手、价差、冲击、停牌与退出规则，并保留未参与选择的末端样本。
\end{enumerate}
